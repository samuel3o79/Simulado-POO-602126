  public class Produto {
      public String nome;
      private static double preco;
      final int codigo;
  
      public Produto(String nome) {
          codigo = "A100";  // O parâmetro codigo é criado como inteiro e nessa linha de codigo ele recebe uma string, isso gera um erro de compilação
          nome = nome;
      }
 
     public static void aplicarDesconto(double porcentagem) {
         preco = preco - (preco * (porcentagem / 100));
         System.out.println("Produto: " + nome); //nome é atributo de instância, o método estatico não pode acessar atributo de instância
     }
 
     public void setCodigo(int novoCodigo) {
         this.codigo = novoCodigo; //codigo é final então não pode ser reatribuido depois de ser inicializado no construtor
     }
 
     public void getPreco() {
         return preco;//O método é declarado como void e possui um return
     }
 
     public static void main(String[] args) {
         Produto p1 = new Produto("Teclado");
         p1.preco = 150.0;
         p1.codigo = 500;//codigo é final, não pode ser alterado após a construção do objeto
         
         double valor = p1.getPreco();
         System.out.println("Valor: " + valor);
         
         Produto.aplicarDesconto(10);
     }
 }
